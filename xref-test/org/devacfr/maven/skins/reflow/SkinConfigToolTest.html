59    @Test
60    public void testIsActiveLink() throws Exception {
61      ReflowContext context = createSkinConfig();
62      SkinConfigTool skinConfig = context.getSkinConfig();
63      ToolContext velocityContext = context.getVelocityContext();
64      velocityContext.put("alignedFileName", "index.html");
65      assertEquals(true, skinConfig.isActiveLink("."), "should be active");
66      assertEquals(true, skinConfig.isActiveLink(""), "should be active");
67  
68      velocityContext.put("alignedFileName", "summary.html");
69      assertEquals(false, skinConfig.isActiveLink(null), "should be not active");
70      assertEquals(false, skinConfig.isActiveLink("."), "should be not active");
71      assertEquals(false, skinConfig.isActiveLink("index.html"), "should be not active");
72    }

74    @Test
75    public void testIsExternalLink() throws Exception {
76      ReflowContext context = createSkinConfig();
77      SkinConfigTool skinConfig = context.getSkinConfig();
78      assertEquals(true, skinConfig.isExternalLink("http://foo.com"), "should be external");
79      assertEquals(true, skinConfig.isExternalLink("https://foo.com"), "should be external");
80      assertEquals(true, skinConfig.isExternalLink("ftp://foo.com"), "should be external");
81      assertEquals(true, skinConfig.isExternalLink("mailto:john@foo.com"), "should be external");
82      assertEquals(true, skinConfig.isExternalLink("file://foo.com"), "should be external");
83  
84      assertEquals(false, skinConfig.isExternalLink(null), "should be internal");
85      assertEquals(false, skinConfig.isExternalLink("summary.html"), "should be internal");
86      assertEquals(false,
87        skinConfig.isExternalLink("https://devacfr.github.io/reflow-maven-skin/dev/summary.html"),
88        "should be internal");
89    }

150   private ReflowContext createSkinConfig() throws Exception {
151     SkinConfigTool skinConfig;
152     final EasyFactoryConfiguration config = new EasyFactoryConfiguration(false);
153     config.toolbox(Scope.APPLICATION).tool(HtmlTool.class);
154 
155     final ToolManager manager = new ToolManager(false, false);
156     manager.configure(config);
157     ToolContext velocityContext = manager.createContext();
158     final SiteXpp3Reader reader = new SiteXpp3Reader();
159     SiteModel siteModel = null;
160     try (final InputStream in = getResource("default.site.xml").openBufferedStream()) {
161       siteModel = reader.read(in);
162     }
163     ValueParser valueParser = new ValueParser(
164         ImmutableMap.<String, Object> builder().put("velocityContext", velocityContext).build());
165 
166     final MavenProject maven = new MavenProject();
167     maven.setArtifactId("maven-reflow-plugin");
168     maven.setName("Reflow Maven Skin Project");
169     maven.setUrl("https://devacfr.github.io/reflow-maven-skin/");
170 
171     velocityContext.put("project", maven);
172     velocityContext.put("site", siteModel);
173     velocityContext.put("render", new RenderTool());
174     velocityContext.put("currentFileName", "summary.html");
175     velocityContext.put("alignedFileName", "summary.html");
176     skinConfig = new SkinConfigTool();
177     skinConfig.configure(valueParser);
178     return ReflowContext.builder()
179         .maven(maven)
180         .velocityContext(velocityContext)
181         .skinConfig(skinConfig)
182         .siteModel(siteModel)
183         .build();
184   }
